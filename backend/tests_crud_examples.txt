# 1. TRIPS (VOYAGES)
# Endpoint: /trips
# ------------------------------------------------------------------------------

### 1.1 CREATION (POST /api/trips)

# CAS VALIDE : Création d'un voyage complet
# Header: Content-Type: application/json
{
    "title": "Aventure dans les Alpes",
    "destination": {
        "latitude": 45.8326,
        "longitude": 6.8650,
        "address": "Mont Blanc, France"
    },
    "startDate": "2024-07-15T08:00:00Z",
    "endDate": "2024-07-22T18:00:00Z",
    "difficulty": "MODERATE",
    "totalBudget": 1200.50,
    "status": "PLANNED",
    "userId": "user-123"
}

# CAS INVALIDE (Contrôle Saisie) : Titre trop court (min 3)
{
    "title": "Al",
    "startDate": "2024-07-15T08:00:00Z",
    "endDate": "2024-07-22T18:00:00Z",
    "difficulty": "EASY",
    "totalBudget": 500.00,
    "status": "PLANNED",
    "userId": "user-123"
}
# Erreur attendue: "Title must be between 3 and 100 characters"

# CAS INVALIDE (Contrôle Saisie) : Budget négatif
{
    "title": "Voyage Gratuit",
    "startDate": "2024-07-15T08:00:00Z",
    "endDate": "2024-07-22T18:00:00Z",
    "difficulty": "EASY",
    "totalBudget": -10.00,
    "status": "PLANNED",
    "userId": "user-123"
}
# Erreur attendue: "Budget must be a positive number"

# CAS INVALIDE (Contrôle Saisie) : Champs obligatoires manquants
{
    "title": "Voyage Incomplet"
}
# Erreur attendue: "Start date is required", "End date is required", "Difficulty level is required", etc.

### 2.1 CREATION (POST /api/transports)

# CAS VALIDE : Ajout d'un transport
{
    "tripId": "trip-abc-123",
    "mode": "CAR",
    "cost": 85.00,
    "duration": 120,
    "provider": "Sixt Car Rental"
}

# CAS INVALIDE (Contrôle Saisie) : Mode de transport manquant
{
    "tripId": "trip-abc-123",
    "cost": 50.00,
    "duration": 60,
    "provider": "Local Bus"
}
# Erreur attendue: "Mode of transport is required"

# CAS INVALIDE (Contrôle Saisie) : Durée négative ou nulle (doit être positive)
{
    "tripId": "trip-abc-123",
    "mode": "TRAIN",
    "cost": 45.00,
    "duration": 0,
    "provider": "SNCF"
}
# Erreur attendue: "Duration must be positive"

# ------------------------------------------------------------------------------
# 4. ASSOCIATIONS & ASSIGNMENTS (AFFECTATIONS)
# ------------------------------------------------------------------------------

### 4.1 TRIP & TRANSPORT
# Affecter un transport à un voyage
# POST /api/trips/{tripId}/assign-transport/{transportId}
# Exemple:
# POST http://localhost:8080/api/trips/trip-123/assign-transport/trans-456

### 4.2 TRIP & ITINERARY
# Affecter un itinéraire à un voyage
# POST /api/trips/{tripId}/assign-itinerary/{itineraryId}

### 4.3 ITINERARY COMPONENTS
# Affecter une activité à un itinéraire
# POST /api/itineraries/{itineraryId}/assign-activity/{activityId}

# Affecter un POI à un itinéraire
# POST /api/itineraries/{itineraryId}/assign-poi/{poiId}

# Affecter un segment à un itinéraire
# POST /api/itineraries/{itineraryId}/assign-segment/{segmentId}

### 4.4 SEGMENT & SAFETY
# Affecter une alerte à un segment
